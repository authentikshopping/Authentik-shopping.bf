function commander() { if (cart.length === 0) { alert("Votre panier est vide."); return; } const nom = document.getElementById("nom").value.trim(); const tel = document.getElementById("tel").value.trim(); if (!nom || !tel) { alert("Veuillez entrer votre nom et votre téléphone."); return; } const total = cart.reduce((s, p) => s + p.price * p.qty, 0); let message = `🛍️ *Commande Authentik Shopping*\n\n`; message += `👤 Nom : ${nom}\n`; message += `📞 Téléphone : ${tel}\n\n`; message += `📦 Articles :\n`; cart.forEach((item, i) => { message += `${i + 1}. ${item.name} x${item.qty} - ${(item.price * item.qty).toLocaleString()} FCFA\n`; }); message += `\n💰 Total : ${total.toLocaleString()} FCFA\n\n`; const ouvrirWhatsApp = (texte) => { const numero = "22651918847"; // Mets ton numéro ici const url = `https://wa.me/${numero}?text=${encodeURIComponent(texte)}`; window.location.href = url; }; if (navigator.geolocation) { navigator.geolocation.getCurrentPosition( (position) => { const localisation = `📍 Localisation : https://maps.google.com/?q=${position.coords.latitude},${position.coords.longitude}`; ouvrirWhatsApp(message + localisation); }, () => { ouvrirWhatsApp( message + "📍 Le client n'a pas autorisé la localisation." ); }, { enableHighAccuracy: true, timeout: 10000 } ); } else { ouvrirWhatsApp(message + "📍 Localisation indisponible."); } }
